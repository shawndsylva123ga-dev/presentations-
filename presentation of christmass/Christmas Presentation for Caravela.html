import numpy as np

class ChristmasAntiGravityField:
    """Vectorized 2D Anti-Gravity Buoyancy & Cursor Vortex Field."""
    def __init__(self, n_ornaments=65, lift=-0.85, radius=190.0):
        self.pos = np.random.uniform([0, 0], [1920, 1080], (n_ornaments, 2))
        self.vel = np.zeros((n_ornaments, 2))
        self.lift = lift          # Negative gravity => upward harmonic buoyancy
        self.radius = radius

    def step(self, cursor_xy, mode="repel", dt=0.016):
        delta = self.pos - np.asarray(cursor_xy)
        dist = np.linalg.norm(delta, axis=1, keepdims=True) + 1e-5
        mask = (dist < self.radius).astype(float)
        falloff = (1.0 - dist / self.radius) * mask

        radial = delta / dist
        tangential = np.hstack([-radial[:, 1:2], radial[:, 0:1]])
        force = radial * (420.0 if mode == "repel" else -160.0) + tangential * 280.0

        self.vel = self.vel * 0.94 + (force * falloff + [0.0, self.lift * 35.0]) * dt
        self.pos = (self.pos + self.vel * dt) % [1920, 1080]
        return self.pos
